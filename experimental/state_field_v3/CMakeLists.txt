cmake_minimum_required(VERSION 3.16)
project(rbe_closed_frame LANGUAGES CXX)
add_library(rbe_closed_frame SHARED src/frame_api.cpp)
target_compile_features(rbe_closed_frame PUBLIC cxx_std_20)
target_include_directories(rbe_closed_frame PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(rbe_closed_frame PRIVATE -O3 -Wall -Wextra)
endif()
set_target_properties(rbe_closed_frame PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(frame_test tests/frame_test.cpp)
  target_link_libraries(frame_test PRIVATE rbe_closed_frame)
  add_test(NAME frame_test COMMAND frame_test)
endif()
